DROP TABLE ENCOMENDA;
DROP TABLE CAMINHAO;
DROP TABLE LOJA;
DROP TABLE ARMAZEM;

CREATE TABLE Caminhao 
( 
 Cod_Caminhao NUMBER GENERATED BY DEFAULT AS IDENTITY,  
 PesoMax NUMBER(15,2),  
 VolumeMax NUMBER(15,2),  
 NomeMotorista VARCHAR2(30),  
 Placa VARCHAR2(10),
 CONSTRAINT PK_CAMINHAO PRIMARY KEY (Cod_Caminhao)
);

CREATE TABLE Armazem 
( 
 Cod_Armazem NUMBER GENERATED BY DEFAULT AS IDENTITY,  
 Endereco VARCHAR2(50),  
 Contato VARCHAR2(20),  
 Capacidade NUMBER(15,2), -- 1234567890123,45
 CONSTRAINT PK_ARMAZEM PRIMARY KEY (Cod_Armazem)
);

CREATE TABLE Loja 
( 
 Cod_Loja NUMBER GENERATED BY DEFAULT AS IDENTITY,  
 Endereco VARCHAR2(50),  
 Nome VARCHAR2(30),  
 Contato VARCHAR2(20),
  CONSTRAINT PK_LOJA PRIMARY KEY (Cod_Loja)
);

CREATE TABLE Encomenda 
( 
 Numero NUMBER GENERATED BY DEFAULT AS IDENTITY,  
 Numero_Viagem NUMBER,  
 idCaminhao NUMBER REFERENCES CAMINHAO(Cod_Caminhao),  
 idArmazem NUMBER REFERENCES LOJA(Cod_Loja),  
 idLoja NUMBER,  
 Peso NUMBER(15,2),  
 Volume NUMBER(15,2),  
 Descricao VARCHAR2(50),  
 Valor NUMBER(15,2),  
 DataEncomenda DATE,  
 horario TIMESTAMP,  
 Rota VARCHAR2(40),
 CONSTRAINT PK_ENCOMENDA PRIMARY KEY (Numero),
 CONSTRAINT FK_ARMAZEM FOREIGN KEY (idArmazem) REFERENCES ARMAZEM(Cod_Armazem)
);
